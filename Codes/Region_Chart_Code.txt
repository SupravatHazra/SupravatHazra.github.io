Option Explicit
Sub UpdateRegionSummary()

    '==========================================================
    ' Procedure : UpdateRegionSummary
    ' Purpose   : Creates the Region Summary Table
    '==========================================================

    Dim wsOutput As Worksheet
    Dim wsDash As Worksheet

    Dim Dict As Object

    Dim LastRow As Long
    Dim i As Long
    Dim r As Long

    Dim Region As String
    Dim Sales As Double

    Dim Key As Variant

    Set wsOutput = Worksheets("Output")
    Set wsDash = Worksheets("Dashboard")

    LastRow = wsOutput.Cells(wsOutput.Rows.Count, "A").End(xlUp).Row

    wsDash.Range("AD2:AE100").ClearContents

    If LastRow < 2 Then Exit Sub

    wsDash.Range("AD2").Value = "Region"
    wsDash.Range("AE2").Value = "Total Sales"

    Set Dict = CreateObject("Scripting.Dictionary")

    For i = 2 To LastRow
        Region = wsOutput.Cells(i, "D").Value
        Sales = wsOutput.Cells(i, "E").Value
        If Dict.Exists(Region) Then
            Dict(Region) = Dict(Region) + Sales
        Else
            Dict.Add Region, Sales
        End If
    Next i

    r = 3

    For Each Key In Dict.Keys

        wsDash.Cells(r, "AD").Value = Key
        wsDash.Cells(r, "AE").Value = Dict(Key)

        r = r + 1

    Next Key
    'Format Header
    With wsDash.Range("AD2:AE2")
        .Font.Bold = True
        .Interior.Color = RGB(0, 112, 192)
        .Font.Color = RGB(255, 255, 255)
    End With
    'Number Format
    With wsDash.Range("AE3:AE" & r - 1)
        .NumberFormat = "#,##,##0"
    End With
    wsDash.Columns("AD:AE").AutoFit

End Sub

Sub SortRegionSummary()

    Dim wsDash As Worksheet
    Dim LastRow As Long

    Set wsDash = Worksheets("Dashboard")

    LastRow = wsDash.Cells(wsDash.Rows.Count, "AD").End(xlUp).Row

    If LastRow < 3 Then Exit Sub

    With wsDash.Sort

        .SortFields.Clear

        .SortFields.Add _
            Key:=wsDash.Range("AE3:AE" & LastRow), _
            SortOn:=xlSortOnValues, _
            Order:=xlDescending, _
            DataOption:=xlSortNormal

        .SetRange wsDash.Range("AD2:AE" & LastRow)

        .Header = xlYes
        .MatchCase = False
        .Orientation = xlTopToBottom

        .Apply

    End With

End Sub

Sub SetupRegionChart()

    '==========================================================
    ' Procedure : SetupRegionChart
    ' Purpose   : Creates the Region Chart only once.
    '==========================================================

    Dim wsDash As Worksheet
    Dim ChObj As ChartObject
    Dim LastRow As Long

    Set wsDash = Worksheets("Dashboard")

    LastRow = wsDash.Cells(wsDash.Rows.Count, "AD").End(xlUp).Row

    If LastRow < 3 Then Exit Sub

    On Error Resume Next
    Set ChObj = wsDash.ChartObjects("RegionChart")
    On Error GoTo 0

    If Not ChObj Is Nothing Then Exit Sub

    Set ChObj = wsDash.ChartObjects.Add( _
                    Left:=wsDash.Range("E35").Left, _
                    Top:=wsDash.Range("E35").Top, _
                    Width:=700, _
                    Height:=350)

    ChObj.Name = "RegionChart"

    With ChObj.Chart

        .ChartType = xlColumnClustered

        .SetSourceData Source:=wsDash.Range("AD2:AE" & LastRow)
        .PlotVisibleOnly = False

        .HasTitle = True
        .ChartTitle.Text = "Sales by Region"

        With .ChartTitle.Format.TextFrame2.TextRange.Font
            .Size = 16
            .Bold = msoTrue
        End With

        .HasLegend = False

        .Axes(xlCategory).HasTitle = True
        .Axes(xlCategory).AxisTitle.Text = "Region"

        .Axes(xlCategory).TickLabels.Font.Size = 10
        .Axes(xlCategory).TickLabels.Orientation = 45

        .Axes(xlValue).HasTitle = True
        .Axes(xlValue).AxisTitle.Text = "Total Sales"

        .Axes(xlValue).TickLabels.Font.Size = 10
        .Axes(xlValue).TickLabels.NumberFormat = "#,##,##0"

        .ApplyDataLabels

        With .SeriesCollection(1).DataLabels
            .ShowValue = True
            .Position = xlLabelPositionOutsideEnd
            .Font.Bold = True
            .Font.Size = 10
        End With

        With .SeriesCollection(1).Format.Fill
            .Visible = msoTrue
            .Solid
            .ForeColor.RGB = RGB(0, 176, 80)
        End With

        .ChartGroups(1).GapWidth = 50

        .ChartArea.Format.Line.Visible = msoFalse
        .PlotArea.Format.Line.Visible = msoFalse

        .Axes(xlValue).MajorGridlines.Format.Line.ForeColor.RGB = RGB(220, 220, 220)

    End With

End Sub

Sub UpdateRegionChart()

    Dim wsDash As Worksheet
    Dim ChObj As ChartObject
    Dim LastRow As Long

    Set wsDash = Worksheets("Dashboard")

    LastRow = wsDash.Cells(wsDash.Rows.Count, "AD").End(xlUp).Row

    Set ChObj = wsDash.ChartObjects("RegionChart")

    If LastRow < 3 Then

        On Error Resume Next
        ChObj.Chart.SeriesCollection(1).Delete
        On Error GoTo 0

        Exit Sub

    End If

    With ChObj.Chart

        .ChartType = xlColumnClustered
        .SetSourceData Source:=wsDash.Range("AD2:AE" & LastRow)
        .Refresh

    End With

End Sub

Sub UpdateRegionDashboard()

    UpdateRegionSummary

    SortRegionSummary

    SetupRegionChart

    UpdateRegionChart

End Sub
